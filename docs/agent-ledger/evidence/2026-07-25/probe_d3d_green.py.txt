REG = json.loads(
    (Path("docs/validation/2026-07-21-qb-1-study-registration.json")).read_text()
)
HASH = build_registration(REG)["sha256"]
MODEL_FOLDS = tuple(range(2018, 2026))
H5_FOLDS = (2021, 2022, 2023, 2024)
RESULTS: list[tuple[str, bool, str]] = []


def check(name, condition, detail=""):
    RESULTS.append((name, bool(condition), detail))


def spec(cid):
    return copy.deepcopy(next(r for r in REG["contrasts"] if r["id"] == cid))


def lane(name, season, ids, order):
    rows = []
    for index, pid in enumerate(ids):
        y = float(index + 1)
        pred = y if order == "perfect" else (-y if order == "reverse" else 1.0)
        row = {
            "player_id": pid,
            "target_season": season,
            "y_pred": pred,
            "y_true": y,
            "decision_supported": False,
        }
        if name == "h5":
            row["source_provenance"] = "dynastyprocess_ecr_2qb"
        rows.append(row)
    out = {
        "test_season": season,
        "lane": name,
        "n_predicted": len(rows),
        "predictions": rows,
        "decision_supported": False,
    }
    if name == "h5":
        out["lane_source"] = "dynastyprocess_ecr_2qb"
    return out


def fold(cid, season, n=20, left="perfect", right="perfect"):
    s = spec(cid)
    ids = [f"p{i:03d}" for i in range(n)]
    return score_comparisons(
        lane(s["left"], season, ids, left),
        lane(s["right"], season, ids, right),
        contrast=s,
    )


def contrast(cid, per_fold):
    s = spec(cid)
    return {
        "id": s["id"],
        "lane": s["lane"],
        "left": s["left"],
        "right": s["right"],
        "registered_direction": s["direction"],
        "per_fold": per_fold,
        "folds_present": len(per_fold),
        "decision_supported": False,
    }


def signal():
    per = [
        fold("c01", 2018, 20, "perfect", "reverse"),
        fold("c01", 2019, 20, "perfect", "perfect"),
    ]
    per.extend(fold("c01", s, 0) for s in MODEL_FOLDS[2:])
    return contrast("c01", per)


def reason(fn):
    try:
        fn()
    except QBValidationFailure as exc:
        return exc.reason
    except Exception as exc:  # noqa: BLE001 - probe wants the raw class
        return f"UNNAMED:{type(exc).__name__}"
    return "NO-RAISE"


# 1 — contrast-order independence: c01 seeded by REGISTERED id, not position.
solo = cluster_bootstrap_distribution(signal(), registration=REG)
check(
    "order-independent seed material",
    solo["seed_material"] == [20260716, 0, 1],
    str(solo["seed_material"]),
)

# 2 — permutation determinism across repeated calls.
p1 = cluster_permutation_p(signal(), registration=REG)
p2 = cluster_permutation_p(signal(), registration=REG)
check("permutation deterministic", p1 == p2)

# 3 — p_two can never be 0 (the registered +1 / B+1 correction).
check("p_two floor is 1/(B+1)", p1["p_two"] >= 1 / 10_001, str(p1["p_two"]))

# 4 — bca first-arg strictness (v9 §1): list and the record itself are misuse.
check(
    "bca refuses list",
    reason(lambda: bca_interval([0.1, 0.2], observed=0.1, jackknife=(0.0, 1.0)))
    == "UNNAMED:TypeError",
)
check(
    "bca refuses distribution record",
    reason(lambda: bca_interval(solo, observed=0.1, jackknife=(0.0, 1.0)))
    == "UNNAMED:TypeError",
)

# 5 — empty distribution is unavailable, never a fabricated interval.
empty_iv = bca_interval((), observed=0.1, jackknife=(0.0, 1.0))
check(
    "empty distribution unavailable",
    empty_iv["state"] == "distribution_unavailable"
    and empty_iv["ci95_low"] is None,
)

# 6 — NaN / inf evidence refuses by name on every numeric field.
for field in ("left_pred", "right_pred", "y_true"):
    for bad in (math.nan, math.inf, -math.inf):
        c = signal()
        c["per_fold"][0]["paired_evidence"][0][field] = bad
        check(
            f"non-finite {field} {bad}",
            reason(lambda c=c: build_cluster_universe(c)) == "non_finite_evidence",
        )

# 7 — a plain str SUBCLASS identity key is refused (equality shadowing).
class SubStr(str):
    pass


c = signal()
c["per_fold"][0]["paired_evidence"][0]["player_id"] = SubStr("p000")
check(
    "str-subclass identity key refused",
    reason(lambda: build_cluster_universe(c)) == "non_plain_identity_key",
)

# 8 — finite extremes must not overflow; rank statistics are scale-free.
c = signal()
for row in c["per_fold"][0]["paired_evidence"]:
    row["left_pred"] = row["left_pred"] * 1e308
extreme = pool_paired_deltas(c, registration=REG)
check(
    "1e308 finite extreme pools without overflow",
    extreme["state"] == "ok" and math.isfinite(extreme["pooled_delta"]),
)

# 9 — contradictory record: delta present but pool empty.
c = signal()
c["per_fold"][0]["common_pool_n"] = 0
check(
    "delta with empty pool refused",
    reason(lambda: pool_paired_deltas(c, registration=REG))
    == "fold_record_inconsistent",
)

# 10 — a UserDict contrast is admitted (mapping subclass, not a hostile object).
c = signal()
check(
    "UserDict contrast admitted",
    pool_paired_deltas(UserDict(c), registration=REG)["state"] == "ok",
)

# 11 — BH monotonicity and the all-ones family.
adj = benjamini_hochberg(
    {f"c{i:02d}": i / 14 for i in range(1, 15)}, q=0.10, family_size=14
)
values = [adj["adjusted"][f"c{i:02d}"] for i in range(1, 15)]
check("BH adjusted is monotone non-decreasing", all(
    a <= b + 1e-15 for a, b in zip(values, values[1:])
))
ones = benjamini_hochberg(
    {f"c{i:02d}": 1.0 for i in range(1, 15)}, q=0.10, family_size=14
)
check(
    "all p=1 rejects nothing and caps at 1.0",
    not any(ones["rejected"].values())
    and all(v == 1.0 for v in ones["adjusted"].values()),
)

# 12 — NI sign discipline: delta0 sign is load-bearing, not cosmetic.
dist = (-0.10, 0.0, 0.10, 0.20)
neg = shifted_null_ni_p(dist, observed=0.10, delta0=-0.05)
pos = shifted_null_ni_p(dist, observed=0.10, delta0=0.05)
check("delta0 sign changes p_ni", neg["p_ni"] != pos["p_ni"],
      f"{neg['p_ni']} vs {pos['p_ni']}")

# 13 — orchestrator refuses contrast-order drift.

    "registration_hash": HASH,
    "contrasts": [
        contrast(
            cid,
            [

    ],
    "decision_supported": False,

    shuffled["contrasts"][1],

    reason(

            shuffled, registration=REG, expected_registration_hash=HASH

    == "contrast_set_drift",
)

# 14 — no support_status anywhere in a full run (P9), and no fabricated pooled 0.0.
